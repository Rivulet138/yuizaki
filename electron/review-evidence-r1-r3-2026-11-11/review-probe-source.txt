// Review evidence only. Reuses the existing regression fixture without changing it.
describe('R1-R3 review probes', () => {
  it('CONFIRMS a credential-only failed patch loses its recovery identity when deleted by another panel', async () => {
    state.hydratePreservingDraft(state.captureFormDraft(), saved)
    vi.mocked(settingsClient.save).mockRejectedValueOnce(new Error('credential save failed'))
    state.saveAsrField('api_key', 'synthetic-review-key')
    expect(await state.flushPendingSave()).toBe(false)
    const secondScope = effectScope()
    try {
      const second: any = secondScope.run(() => (SettingsPanel as any).setup({}, { expose: vi.fn() }))
      second.settings.value = structuredClone(saved)
      second.hydratePreservingDraft(second.captureFormDraft(), saved)
      vi.mocked(settingsClient.deleteSetting).mockImplementationOnce(async () => {
        saved.asr.api_key = ''
        return { status: 'ok' }
      })
      await second.clearAsrApiKey()
      await nextTick()
      const observed = {
        failedPatch: settingsSaveState.failed.value,
        pendingPatch: settingsSaveState.pending.value,
        failedRecoveries: settingsSaveState.failedRecoveries.size,
        originalError: state.updateRequest.error,
        originalLabel: state.saveStatusLabel.value,
        originalKey: state.form.asr.api_key,
        deleteCalls: vi.mocked(settingsClient.deleteSetting).mock.calls.length,
      }
      console.log('CREDENTIAL_DELETE_RECOVERY', JSON.stringify(observed))
      expect(observed.failedPatch).toBeNull()
      expect(observed.pendingPatch).toBeNull()
      expect(observed.originalKey).toBe('')
      expect(observed.deleteCalls).toBe(1)
      expect(observed.failedRecoveries).toBe(1)
      expect(observed.originalError).toContain('credential save failed')
      expect(observed.originalLabel).toBe(t('settings.status.saveFailed'))
      second.retryFailedSave()
      await second.waitForSaveIdle()
      expect(state.updateRequest.error).toContain('credential save failed')
    } finally { secondScope.stop() }
  })

  it('CHECKS a pre-existing panel returning an unsubmitted slider draft to baseline during another save', async () => {
    state.hydratePreservingDraft(state.captureFormDraft(), saved)
    const secondScope = effectScope()
    const response = deferred<any>()
    try {
      const second: any = secondScope.run(() => (SettingsPanel as any).setup({}, { expose: vi.fn() }))
      second.settings.value = structuredClone(saved)
      second.hydratePreservingDraft(second.captureFormDraft(), saved)
      const baseline = second.form.llm.temperature
      vi.mocked(settingsClient.save).mockReturnValueOnce(response.promise)
      state.form.llm.temperature = 1.2
      state.saveLlmField('temperature', 1.2)
      const saving = state.flushPendingSave()
      second.updateLlmParameter('temperature', baseline === 0.8 ? 0.9 : 0.8)
      second.updateLlmParameter('temperature', baseline)
      response.resolve({ status: 'ok', normalized_settings: { ...saved, llm: { ...saved.llm, temperature: 1.2, profiles: { custom: { ...saved.llm, temperature: 1.2 } } } } })
      await saving
      console.log('BASELINE_SLIDER_DRAFT', JSON.stringify({ baseline, form: second.form.llm.temperature, settings: second.settings.value.llm.temperature }))
      expect(second.form.llm.temperature).toBe(1.2)
    } finally { response.resolve({ status: 'ok', normalized_settings: saved }); secondScope.stop() }
  })
})
